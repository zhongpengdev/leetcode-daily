```c
int Find(int x) {
    while(parent[x] != x) {
        x = parent[x];
    }

    return x;
}
```

```c
int Union(int x, int y) {
    // find the value's root
    int rx = Find(x);
    int ry = Find(y);

    // change rx's parent array value into ry 
    if(rx != ry) {
        parent[rx] = ry;
    }
}
```
